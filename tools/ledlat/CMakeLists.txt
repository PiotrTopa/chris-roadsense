cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(ledlat C CXX ASM)
pico_sdk_init()
add_executable(ledlat main.c)
pico_generate_pio_header(ledlat ${CMAKE_CURRENT_LIST_DIR}/ws2812.pio)
target_link_libraries(ledlat pico_stdlib hardware_pio pico_bootrom)
pico_enable_stdio_usb(ledlat 1)
pico_enable_stdio_uart(ledlat 0)
pico_add_extra_outputs(ledlat)
