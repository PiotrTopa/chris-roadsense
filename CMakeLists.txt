cmake_minimum_required(VERSION 3.16)
project(roadsense CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Built on the phone itself (Snapdragon 845: Cortex-A75 + A55, ARMv8.2-A).
option(ROADSENSE_NATIVE "Tune for the build machine CPU (-mcpu=native)" ON)
include(CheckCXXCompilerFlag)
if(ROADSENSE_NATIVE)
  check_cxx_compiler_flag(-mcpu=native HAVE_MCPU_NATIVE)
  if(HAVE_MCPU_NATIVE)
    add_compile_options(-mcpu=native)
  endif()
endif()
add_compile_options(-Wall -Wextra -O3)

find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)

add_subdirectory(camerad)
# camerad-dvr: alternative camera source (USB DVR camera 1902:3241, MJPEG over mass storage).
find_package(JPEG)
include(CMakeDependentOption)
cmake_dependent_option(ROADSENSE_CAMERAD_DVR "Build camerad-dvr (needs libjpeg-turbo)" ON "JPEG_FOUND" OFF)
if(ROADSENSE_CAMERAD_DVR)
  add_subdirectory(camerad-dvr)
endif()
# camerad-replay: alternative camera source (a recorded ride, MJPEG file, looping).
cmake_dependent_option(ROADSENSE_CAMERAD_REPLAY "Build camerad-replay (needs libjpeg-turbo)" ON "JPEG_FOUND" OFF)
if(ROADSENSE_CAMERAD_REPLAY)
  add_subdirectory(camerad-replay)
endif()
add_subdirectory(modeld)
add_subdirectory(envd)
add_subdirectory(laned)
add_subdirectory(tools)
